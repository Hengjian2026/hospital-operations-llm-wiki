# 核心内容摘要

医保结算单数据是医院病种运营的常用数据，是医院进行盈余分析的最直接的数据来源。根据本地医保病种的得分情况可以对病人进行各种分类，如确定医保类型、倍率情况、病种入组情况等。在日常统计分析总，我们常需要从汇总到明细的成成下钻，实现数据的溯源。其中旭日图是一种可以展示数据分层、渐渐明细的图形之一。
1.旭日图简介

旭日图相当于多个饼图的组合，但饼图只能体现一层数据的比例情况，而旭日图不仅可以体现数据比例，还能体现数据层级之间的关系。旭日图的本质是树状关系，因此也被称为极坐标下的矩形树图。

2.样例数据

表单中对病人入组、医疗付费方式和费用倍率情况。可以用旭日图描述所有记录按照医疗付费方式->倍率情况->DIP名称的构成情况。

# 加载必要的包
library(readxl)
library(dplyr)
library(plotly)
library(viridis)
#附件所在物理地址
setwd("D:/")

# 读取Excel文件
data <- read_excel("样例数据.xlsx", sheet = "数据明细表")

# 数据预处理
# 计算每个DIP名称在每个医疗付费方式和倍率情况下的病例数
summary_data <- data %>%
  group_by(`医疗付费方式`, `倍率情况`, `DIP名称`) %>%
  summarise(病例数 = n(), .groups = 'drop') %>%
  arrange(`医疗付费方式`, `倍率情况`, desc(病例数))

# 创建分层结构数据
# 第一层：医疗付费方式
# 第二层：倍率情况
# 第三层：DIP名称

# 为每个层级创建ID
summary_data <- summary_data %>%
  mutate(
    id = paste(`医疗付费方式`, `倍率情况`, `DIP名称`, sep = "-"),
    parent_id = case_when(
      !is.na(`DIP名称`) & !is.na(`倍率情况`) & !is.na(`医疗付费方式`) ~ 
        paste(`医疗付费方式`, `倍率情况`, sep = "-"),
      !is.na(`倍率情况`) & !is.na(`医疗付费方式`) ~ 
        `医疗付费方式`,
      TRUE ~ ""
    ),
    label = `DIP名称`,
    value = 病例数
  )

# 创建父节点数据
parent_nodes <- summary_data %>%
  group_by(`医疗付费方式`, `倍率情况`) %>%
  summarise(value = sum(病例数), .groups = 'drop') %>%
  mutate(
    id = paste(`医疗付费方式`, `倍率情况`, sep = "-"),
    parent_id = `医疗付费方式`,
    label = `倍率情况`
  )

# 创建根节点数据
root_nodes <- parent_nodes %>%
  group_by(`医疗付费方式`) %>%
  summarise(value = sum(value), .groups = 'drop') %>%
  mutate(
    id = `医疗付费方式`,
    parent_id = "",
    label = `医疗付费方式`
  )

# 合并所有数据
all_nodes <- bind_rows(
  root_nodes %>% select(id, parent_id, label, value),
  parent_nodes %>% select(id, parent_id, label, value),
  summary_data %>% select(id, parent_id, label, value)
)
